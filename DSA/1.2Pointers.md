1. Two Sum 
```python
def two_sum(nums, target):
    seen = {}
    for i, num in enumerate(nums):
        complement = target - num
        if complement in seen:
            return [seen[complement], i]
        seen[num] = i

## Using 2 pointer 

def sum_2_sorted(nums, target):
    left, right = 0, len(nums) - 1
    while left < right:
        curr_sum = nums[left] + nums[right]
        if curr_sum == target:
            return [left, right]
        elif curr_sum < target:
            left += 1
        else:
            right -= 1
    return []
```

2. 3Sum
How it works:
- Sort once — O(n log n).
- Loop i over each element as the "anchor"; for each anchor, run the 2-pointer scan on the rest of the array for target - nums[i].
- Skip duplicate values at all three positions to avoid duplicate triplets in the result.
```python 
def three_sum(nums, target=0):
    nums.sort()
    result = []
    n = len(nums)

    for i in range(n - 2):
        # skip duplicate anchors
        if i > 0 and nums[i] == nums[i - 1]:
            continue

        left, right = i + 1, n - 1
        remaining = target - nums[i]

        while left < right:
            curr_sum = nums[left] + nums[right]

            if curr_sum == remaining:
                result.append([nums[i], nums[left], nums[right]])
                left += 1
                right -= 1
                # skip duplicates for left/right
                while left < right and nums[left] == nums[left - 1]:
                    left += 1
                while left < right and nums[right] == nums[right + 1]:
                    right -= 1
            elif curr_sum < remaining:
                left += 1
            else:
                right -= 1

    return result



## Approach 2 
def three_sum_hashset(nums, target=0):
    nums.sort()  # still sort to make dedup easy
    result = []
    n = len(nums)

    for i in range(n - 2):
        if i > 0 and nums[i] == nums[i - 1]:
            continue

        seen = set()
        remaining = target - nums[i]

        for j in range(i + 1, n):
            complement = remaining - nums[j]
            if complement in seen:
                triplet = [nums[i], complement, nums[j]]
                if triplet not in result:  # avoid dup triplets
                    result.append(triplet)
            seen.add(nums[j])

    return result
```


3. Container With Most Water:
```python
def most_water(heights):
    max_water = 0
    left = 0
    right = len(heights) - 1
    while left < right:
        max_water = max(max_water, min(heights[right],heights[left]) * (right - left) )

        if heights[left] <= heights[right]: 
            left = left + 1
        else:
            right = right - 1

    return max_water
```
 
 4. Remove Duplicates from Sorted Array
 ```python 
 def rmv_duplicates(arr:list) -> list:
    removed_duplicated = []            
    one_pointer = 0
    second_pointer = one_pointer + 1
    for i in range(0, len(arr) - 1):
        if arr[one_pointer] == arr[second_pointer]:
            second_pointer += 1
        else: 
            removed_duplicated.append(arr[one_pointer])
            one_pointer = second_pointer
            second_pointer += 1
            
    removed_duplicated.append(arr[len(arr) - 1])
    return removed_duplicated
 ```

 5. Palindrome: 
 ```python
 def isPalindrome(s: str) -> bool:
    def is_alphanumeric(char):
        return ('a' <= char <= 'z') or ('A' <= char <= 'Z') or ('0' <= char <= '9')
     
    i, j = 0, len(s) - 1
    while i < j:
        if not is_alphanumeric(s[i]):
            i += 1
        elif not is_alphanumeric(s[j]):
            j -= 1
        elif s[i].lower() != s[j].lower():
            return False
        else:
            i, j = i + 1, j - 1
    return True
 ```
